/* Queen Usman (author site): palette toggle, matching the novel site. - Night Ink is the default, Ivory Paper is the alternate. - Applies the saved choice before first paint, so there is no flash. - Adds the toggle button to the existing header without touching the app code. */ (function () { var KEY = "qu:theme"; var root = document.documentElement; var BAR_COLOR = { dark: "#11100f", ivory: "#f8f5ee" }; var SUN = ''; var MOON = ''; function read() { try { return localStorage.getItem(KEY); } catch (e) { return null; } } function write(mode) { try { localStorage.setItem(KEY, mode); } catch (e) { /* private mode: ignore */ } } function current() { return root.classList.contains("theme-ivory") ? "ivory" : "dark"; } function paint(btn, mode) { var ivory = mode === "ivory"; btn.setAttribute("aria-pressed", ivory ? "true" : "false"); btn.setAttribute("aria-label", ivory ? "Switch to Night Ink palette" : "Switch to Ivory paper palette"); btn.innerHTML = (ivory ? SUN : MOON) + '' + (ivory ? "Night Ink" : "Ivory Mode") + ""; } function apply(mode) { root.classList.remove("theme-dark", "theme-ivory"); root.classList.add(mode === "ivory" ? "theme-ivory" : "theme-dark"); var meta = document.querySelector('meta[name="theme-color"]'); if (meta) meta.setAttribute("content", BAR_COLOR[mode]); var btn = document.querySelector(".qu-theme-toggle"); if (btn) paint(btn, mode); } function inject() { var bar = document.querySelector("#root > header > .page"); if (!bar || bar.querySelector(".qu-theme-toggle")) return; var btn = document.createElement("button"); btn.type = "button"; btn.className = "qu-theme-toggle"; paint(btn, current()); btn.addEventListener("click", function () { var next = current() === "ivory" ? "dark" : "ivory"; write(next); apply(next); }); var menuBtn = bar.querySelector('button[aria-controls="mobile-menu"]'); bar.insertBefore(btn, menuBtn && menuBtn.parentNode === bar ? menuBtn : null); } function boot() { inject(); var host = document.getElementById("root") || document.body; new MutationObserver(function () { if (!document.querySelector(".qu-theme-toggle")) inject(); }).observe(host, { childList: true, subtree: true }); } // Apply the saved palette immediately apply(read() === "ivory" ? "ivory" : "dark"); // Keep other open tabs in step window.addEventListener("storage", function (e) { if (e.key === KEY) apply(e.newValue === "ivory" ? "ivory" : "dark"); }); if (document.readyState === "loading") { document.addEventListener("DOMContentLoaded", boot); } else { boot(); } })();
